document.getElementById('generateForm').addEventListener('submit', async (e) => { e.preventDefault(); const formData = new FormData(e.target); try { const response = await fetch('http://localhost:5050/api/generate-image', { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ prompt: formData.get('prompt'), width: parseInt(formData.get('width')), height: parseInt(formData.get('height')) }) }); const data = await response.json(); if (data.imageUrl) { const img = document.getElementById('generatedImage'); img.src = data.imageUrl; img.style.display = 'block'; } else { alert('Error: ' + (data.error || 'Failed to generate image')); } } catch (error) { alert('Error: ' + error.message); } });
